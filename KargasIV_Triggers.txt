//============================================================
//  KARGAS IV - Last Frontier (Campaign Co-op Comp Stomp)
//  Trigger Script for SCMDraft 2 TrigEdit Import
//
//  사용법:
//    1. SCMDraft 2 -> Triggers (Ctrl+T)
//    2. File -> Import Text Triggers
//    3. 이 파일 선택
//
//  필요한 Locations (트리거 작성 전 SCMDraft에서 만들어야 함):
//    - "P1 Start", "P2 Start", "P3 Start", "P4 Start" (각 본진 영역)
//    - "P5 Base", "P6 Base", "P7 Base", "P8 Base"   (각 컴 본진 영역)
//    - "Mission2 Civilians" (미션2 난민 시작 지점)
//    - "Mission2 Goal"      (난민 호송 목적지 = 본진)
//    - "Crystal A", "Crystal B", "Crystal C" (미션3 점령 지점 3곳)
//    - "Hero Spawn"         (영웅 등장 위치 = 본진 중앙)
//    - "UED Terminal"       (미션4 점령 건물 영역)
//    - "Boss Area"          (미션5 보스 영역)
//    - "Anywhere"           (전체 맵)
//
//  Death Counters (스위치 대용):
//    Player 8의 가상 유닛 카운터 사용
//    "Mission_State"  값으로 미션 진행 추적
//      0 = 시작 전, 1 = 미션1 진행중, 2 = 클리어, ...
//============================================================


//---------- INITIAL SETUP ----------

Trigger("Player 1","Player 2","Player 3","Player 4","Player 5","Player 6","Player 7","Player 8"){
Conditions:
    Always();

Actions:
    Set Mission Objectives("미션 1: 착륙지를 확보하라 - 북서쪽 저그 군락을 격파하시오");
    Set Countdown Timer(Set To, 0);
    Set Deaths("Current Player", "Terran Marine", Set To, 0);
    Preserve Trigger();
    Comment("초기 설정 - 한 번만 실행");
}

//---------- 시작 메시지 (게임 시작 시) ----------

Trigger("All players"){
Conditions:
    Elapsed Time(At Most, 1);

Actions:
    Display Text Message(Always Display, "\x07=== 최후의 전선: 카르가스 IV ===");
    Display Text Message(Always Display, "\x04당신들의 함선은 변방 행성에 추락했다.");
    Display Text Message(Always Display, "\x04생존을 위해, 그리고 진실을 밝히기 위해...");
    Wait(2000);
    Display Text Message(Always Display, "\x08[미션 1] 착륙지 확보: 북서 저그 군락 파괴");
    Play WAV("sound\\Misc\\BellTrig.wav", 1500);
    Set Deaths("Player 8", "Zerg Zergling", Set To, 1);
    Comment("미션 1 시작 표시 (Mission_State = 1)");
}


//============================================================
//  MISSION 1: 저그 군락 격파 (P5)
//============================================================

Trigger("Player 5"){
Conditions:
    Deaths("Player 8", "Zerg Zergling", Exactly, 1);
    Elapsed Time(At Least, 5);

Actions:
    Run AI Script("Zerg Custom Level");
    Set Deaths("Player 8", "Zerg Zergling", Set To, 2);
    Comment("미션1 AI 시작");
}

Trigger("Player 5"){
Conditions:
    Elapsed Time(At Least, 60);
    Deaths("Player 8", "Zerg Zergling", At Most, 2);

Actions:
    Create Unit(6, "Zerg Zergling", "P1 Start", "Player 5");
    Create Unit(6, "Zerg Zergling", "P4 Start", "Player 5");
    Order(All, "Player 5", "P5 Base", Attack, "P1 Start");
    Display Text Message(Always Display, "\x06[경고] 저그 정찰대가 본진을 공격합니다!");
    Wait(180000);
    Preserve Trigger();
    Comment("저그 웨이브 - 3분마다");
}

Trigger("All players"){
Conditions:
    Deaths("Player 5", "Buildings", Exactly, 0);
    Deaths("Player 8", "Zerg Zergling", At Most, 2);

Actions:
    Display Text Message(Always Display, "\x07[미션 1 완료] 착륙지 확보!");
    Display Text Message(Always Display, "\x04영웅 합류: 짐 레이너가 P1에게 전송되었습니다.");
    Create Unit(1, "Hero Jim Raynor (Vulture)", "P1 Start", "Player 1");
    Play WAV("sound\\Misc\\Hero1.wav", 2000);
    Set Resources("Player 1", Add, 200, ore);
    Set Resources("Player 2", Add, 200, ore);
    Set Resources("Player 3", Add, 200, ore);
    Set Resources("Player 4", Add, 200, ore);
    Set Mission Objectives("미션 2: 잃어버린 형제들 - 남서쪽 난민 10명을 본진까지 호송하라");
    Set Deaths("Player 8", "Zerg Zergling", Set To, 3);
    Comment("미션1 클리어 -> 미션2 시작");
}


//============================================================
//  MISSION 2: 난민 호송
//============================================================

Trigger("All players"){
Conditions:
    Deaths("Player 8", "Zerg Zergling", Exactly, 3);

Actions:
    Display Text Message(Always Display, "\x08[미션 2] 잃어버린 형제들");
    Display Text Message(Always Display, "\x04남서쪽에 추락한 생존자들이 있다. 호송하라.");
    Create Unit(10, "Terran Civilian", "Mission2 Civilians", "Player 1");
    Set Deaths("Player 8", "Zerg Zergling", Set To, 4);
    Set Deaths("Player 8", "Terran Civilian", Set To, 0);
    Run AI Script("Terran Custom Level");
    Comment("난민 10기 생성, UED AI 시작");
}

Trigger("Player 1","Player 2","Player 3","Player 4"){
Conditions:
    Deaths("Player 8", "Zerg Zergling", Exactly, 4);
    Bring("Current Player", At Least, 1, "Terran Civilian", "Mission2 Goal");

Actions:
    Remove Unit At Location(1, "Terran Civilian", "Current Player", "Mission2 Goal");
    Set Deaths("Player 8", "Terran Civilian", Add, 1);
    Display Text Message(Current Player Only, "\x04난민 1명 구조됨");
    Preserve Trigger();
    Comment("난민이 목적지 도달 시 카운트");
}

    Deaths("Player 8", "Terran Civilian", At Least, 7);

    Play WAV("sound\\Misc\\Hero2.wav", 2000);

    Set Deaths("Player 8", "Zerg Zergling", Set To, 5);
    Comment("미션2 클리어");
}
