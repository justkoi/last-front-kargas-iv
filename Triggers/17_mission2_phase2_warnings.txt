// 4분 잔여
Trigger("All Players"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Countdown Timer(At most, 240);
	Countdown Timer(At least, 236);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Display Text Message(Always Display, "[노바] 진입로 차단 임박. 서두르세요, 사령관.");
}
