// 4분 잔여
Trigger("All Players"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Countdown Timer(At most, 240);
	Countdown Timer(At least, 236);
Actions:
	Play WAV("sound\\Misc\\Bell.wav", 1500);
	Display Text Message(Always Display, "[노바] 변종 신호 증가 중. 서두르세요.");
}
