//-----------------------------------------------------------------//
//  미션1 클리어 → 미션2 페이즈1 진입 (Victory() 호출 안 함)
//  Valkyrie DC: 1 → 2, Vulture DC 0 리셋(페이즈 타이머)
//-----------------------------------------------------------------//

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Bring("Player 5", "Zerg Hatchery", "P5 Base", At most, 0);
	Bring("Player 5", "Zerg Lair", "P5 Base", At most, 0);
	Bring("Player 5", "Zerg Hive", "P5 Base", At most, 0);
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
Actions:
	Set Deaths("Player 8", "Terran Valkyrie", Set To, 2);
	Set Deaths("Player 8", "Terran Vulture", Set To, 0);
	Play WAV("sound\\Misc\\TRescue.wav", 2500);
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "  [미션 1 완료] 착륙지 확보 완료");
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "[호크] 첫 거점은 우리 손에 들어왔다. 잘 싸웠다, 선봉대.");
	Display Text Message(Always Display, "[보상] 전 부대 미네랄 +500 / 가스 +200");
	Set Resources("Current Player", Add, 500, ore);
	Set Resources("Current Player", Add, 200, gas);
}

//-----------------------------------------------------------------//
//  패배 - 단계별 가드. 미션1·2 어느 단계든 본진 전멸 시 패배.
//  Valkyrie ≥ 1 (게임 시작 후) + Elapsed Time 가드(전환 보호)
//-----------------------------------------------------------------//

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Command("Current Player", "Buildings", At most, 0);
	Deaths("Player 8", "Terran Valkyrie", At least, 1);
	Elapsed Time(At least, 60);
Actions:
	Play WAV("sound\\Misc\\OutOfGas.wav", 2000);
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "  [작전 실패] 통신 두절");
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "[지휘부] ...선봉대? 응답하라... 응답하라...");
	Wait(1500);
	Defeat();
}
