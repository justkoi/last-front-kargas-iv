//-----------------------------------------------------------------//
//  Mission 2 - P6/P7 uncaptured-colony penalty assaults
//
//  12 DC tick = 1 second. Easy standard cooldown is 1980s = 23760 ticks (33:00).
//  First and repeated cooldowns use the same jitter bands from mission2 entry.
//  Easy jitter bands are 32:15 / 33:00 / 33:45. Roll triggers subtract only the
//  cooldown they consumed, so excess progress from repeated Hatchery/Lair/Hive
//  kills stays queued.
//
//  Waves may fire during mission1 or mission2, but not after victory.
//  They are rolled randomly, then spawned across Spawn1~4 in parallel
//  timed batches to reduce blocked creation space. Spawn timers only count up;
//  separate stage counters prevent repeated or chained creation. Air-heavy
//  waves are split into smaller sub-batches. Spawned groups
//  rally to Waypoint1~5. For 60s after the final batch,
//  P6/P7 assault units reaching the selected waypoint redirect
//  toward the already selected P1~P4 Hunt location.
//-----------------------------------------------------------------//

// Seed the first main-assault wave type at game start so the colony's air/ground
// kind is already decided before any warning fires. This also covers mission1
// penalty assaults (awakened outer colonies) and convenient testing. One-shot:
// no Preserve, so it self-disables after the first pass.
Trigger("Player 8"){
Conditions:
	Switch("Switch132", Not Set);
Actions:
	Set Switch("Switch61", randomize);
	Set Switch("Switch62", randomize);
	Set Switch("Switch65", randomize);
	Set Switch("Switch66", randomize);
	Set Switch("Switch132", set);
}
