//-----------------------------------------------------------------//
//  Mission 2 - P6/P7 uncaptured-colony penalty assaults
//
//  12 DC tick = 1 second. Base cooldown is 1080s = 12960 ticks.
//  Roll triggers subtract only the cooldown they consumed, so excess
//  progress from repeated Hatchery/Lair/Hive kills stays queued.
//  Hatchery/Lair/Hive kills add 80s worth of cooldown progress.
//
//  Waves may fire during mission1 or mission2, but not after victory.
//  They are rolled randomly, then spawned across Spawn1~4 in parallel
//  timed batches to reduce blocked creation space. Spawn timers only count up;
//  separate stage counters prevent repeated or chained creation. Air-heavy
//  waves are split into smaller sub-batches. Spawned groups
//  rally to Waypoint1~5. For 120s after the final batch,
//  P6/P7 assault units reaching the selected waypoint redirect
//  toward the already selected P1~P4 Hunt location.
//-----------------------------------------------------------------//

// Reset assault counters once on mission2 entry.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Vulture", At most, 12);
	Switch("Switch50", Not Set);
Actions:
	Set Switch("Switch26", randomize);
	Set Switch("Switch27", set);
	Set Switch("Switch28", clear);
	Set Switch("Switch29", set);
	Set Switch("Switch30", clear);
	Set Switch("Switch37", clear);
	Set Switch("Switch38", clear);
	Set Switch("Switch39", clear);
	Set Switch("Switch40", clear);
	Set Switch("Switch41", clear);
	Set Switch("Switch48", clear);
	Set Switch("Switch49", clear);
	Set Switch("Switch61", clear);
	Set Switch("Switch62", clear);
	Set Switch("Switch63", clear);
	Set Switch("Switch64", clear);
	Set Switch("Switch65", clear);
	Set Switch("Switch66", clear);
	Set Switch("Switch67", clear);
	Set Switch("Switch68", clear);
	Set Deaths("Player 8", "Terran Ghost", Set To, 0);
	Set Deaths("Player 8", "Zerg Zergling", Set To, 0);
	Set Deaths("Player 8", "Zerg Hydralisk", Set To, 0);
	Set Deaths("Player 8", "Zerg Lurker", Set To, 0);
	Set Deaths("Player 8", "Zerg Mutalisk", Set To, 0);
	Set Deaths("Player 8", "Zerg Ultralisk", Set To, 0);
	Set Deaths("Player 8", "Zerg Devourer", Set To, 0);
	Set Deaths("Player 8", "Zerg Scourge", Set To, 0);
	Set Deaths("Player 8", "Protoss Reaver", Set To, 0);
	Set Deaths("Player 8", "Zerg Guardian", Set To, 0);
	Set Deaths("Player 8", "Protoss Observer", Set To, 0);
}
