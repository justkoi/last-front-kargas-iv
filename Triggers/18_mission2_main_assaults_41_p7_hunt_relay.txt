// P7 random Hunt relay. Final spawn batch 이후 60초 동안 활성.
// P7 병력이 Waypoint에 머무는 동안 1초마다 선택된 P1~P4 Hunt 목적지로 재명령한다.
// 참고: Waypoint -> Hunt Order는 Bring 조건 없이 유닛 타입별 Order를 발행하므로, 해당 위치에 없는 유닛 타입은 영향이 없다.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", At least, 1); //미션 진행 게이트 : 미션1에서 출발한 공세도 릴레이 추적
	Deaths("Player 8", "Terran Valkyrie", At most, 3); //미션 종료 전 게이트 : 클리어 이후 릴레이 중단
	Deaths("Player 8", "Zerg Mutalisk", Exactly, 2); //P7 주공세 상태 : 2일 때 Waypoint -> Hunt 릴레이 단계
	Deaths("Player 8", "Zerg Guardian", At most, 720); //P7 waypoint 창 : final spawn 후 60초(720틱) 이내만 Hunt 릴레이 허용
Actions:
	Set Switch("Switch52", set); //P7 릴레이 enable 스위치 열기 : state=2와 window 조건이 살아 있음을 표시
	Preserve Trigger(); //반복 트리거 유지 : 릴레이 창이 열려 있는 동안 계속 재평가
}
