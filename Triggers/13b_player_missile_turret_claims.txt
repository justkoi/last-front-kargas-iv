//============================================================
//  Player-claimable missile turrets
//  Preplace one Player 9 Terran Missile Turret at each MD1~MD6 location.
//============================================================

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Bring("Current Player", "Men", "MD1", At least, 1);
	Bring("Player 9", "Terran Missile Turret", "MD1", At least, 1);
Actions:
	Modify Unit Hit Points("Player 9", "Terran Missile Turret", 100, 1, "MD1"); //수리 : P9 터렛 체력 100%로 복구 후 양도
	Give Units to Player("Player 9", "Current Player", "Terran Missile Turret", 1, "MD1");
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Minimap Ping("MD1"); //수리 완료 피드백 : 해당 플레이어(Current Player) 트리거 인스턴스만 발동
	Display Text Message(Always Display, "<07>전술 지원: <04>고장난 미사일 터렛을 수리했습니다!");
}
