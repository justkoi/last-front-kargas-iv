// 변형 데스카운트 세팅 - 변종 1회 순회가능하게 7200
Trigger("Player 5"){
Conditions:
	Elapsed Time(At least, 895);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Set To, 7200);
}

// 변형 A — Index 0: 지상 스웜
Trigger("Player 5"){
Conditions:
	Elapsed Time(At least, 900);
	Deaths("Player 8", "Terran Wraith", At least, 1800);
	Command("Player 5", "Zerg Hive", At least, 1);
	Deaths("Player 8", "Terran Dropship", Exactly, 0);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 24, "P5 Base");
	Create Unit("Player 5", "Zerg Hydralisk", 6, "P5 Base");
	Create Unit("Player 5", "Zerg Lurker", 4, "P5 Base");
	Order("Player 5", "Zerg Zergling", "P5 Base", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Base", "Player Destination", Patrol);
	Order("Player 5", "Zerg Lurker", "P5 Base", "Player Destination", Patrol);
	Set Deaths("Player 8", "Terran Wraith", Subtract, 1800);
	Set Deaths("Player 8", "Terran Dropship", Set To, 1);
	Display Text Message(Always Display, "[경보 - 변종 A] 지상 스웜! 럴커 매복 다수!");
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Preserve Trigger();
}
