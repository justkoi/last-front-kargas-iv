//-----------------------------------------------------------------//
//  DefenseCenter1/2 방어 거점 인계 (난이도별)
//
//  쉬움(Academy DC = 1): 미션2 인트로 시점(Vulture >= 756)에 각 거점의
//    모든 P9 무적 건물을 무작위 생존 플레이어에게 통제권 이양(센터별 독립 추첨).
//    이양 시 무적 해제 → 플레이어 방어선으로 활용. 가이드 목적.
//  보통(Academy DC = 2): 난이도 확정 즉시 포토캐논·파일런만 파괴.
//    미션2 인트로에 잔존 건물을 쉬움과 동일하게 무작위 생존 플레이어에게 인계.
//  어려움+(Academy DC >= 3): 난이도 확정 즉시 두 거점 P9 건물 전량 파괴.
//
//  DefenseCenter1/2 에는 에디터에서 P9 소유 건물을 무적 상태로 미리 배치해야 함.
//
//  Switch134/135 : DefenseCenter1 인계 무작위 생존 플레이어 선택자
//  Switch136     : DefenseCenter1 인계 완료(가드)
//  Switch137/138 : DefenseCenter2 인계 무작위 생존 플레이어 선택자
//  Switch139     : DefenseCenter2 인계 완료(가드)
//    선택자 조합: clear/clear=P1, set/clear=P2, clear/set=P3, set/set=P4
//    죽은 플레이어 조합이 뽑히면 give가 발동 안 됨 → 다음 사이클 재추첨.
//-----------------------------------------------------------------//

//----- (A1) 보통 난이도: 난이도 확정 즉시 포토캐논·파일런만 제거 (1회성) -----//
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Academy", Exactly, 2); //보통 난이도 확정
Actions:
	Set Invincibility("Player 9", "Protoss Photon Cannon", "DefenseCenter1", disabled); //파괴 대상만 무적 해제
	Set Invincibility("Player 9", "Protoss Pylon", "DefenseCenter1", disabled);
	Kill Unit At Location("Player 9", "Protoss Photon Cannon", All, "DefenseCenter1");
	Kill Unit At Location("Player 9", "Protoss Pylon", All, "DefenseCenter1");
	Set Invincibility("Player 9", "Protoss Photon Cannon", "DefenseCenter2", disabled);
	Set Invincibility("Player 9", "Protoss Pylon", "DefenseCenter2", disabled);
	Kill Unit At Location("Player 9", "Protoss Photon Cannon", All, "DefenseCenter2");
	Kill Unit At Location("Player 9", "Protoss Pylon", All, "DefenseCenter2");
}

//----- (A2) 어려움+ 난이도: 난이도 확정 즉시 거점 전량 파괴 (1회성) -----//
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Academy", At least, 3); //어려움/매우 어려움 확정
Actions:
	Set Invincibility("Player 9", "Buildings", "DefenseCenter1", disabled); //파괴 전 무적 해제 : 무적 상태면 Kill 불가
	Set Invincibility("Player 9", "Buildings", "DefenseCenter2", disabled);
	Kill Unit At Location("Player 9", "Buildings", All, "DefenseCenter1"); //가이드용 거점 제거
	Kill Unit At Location("Player 9", "Buildings", All, "DefenseCenter2");
}

//----- (B) DefenseCenter1 추첨 (Preserve, 완료 전까지 매 사이클 재추첨) -----//
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Academy", At least, 1);    //쉬움/보통 : 거점 인계
	Deaths("Player 8", "Terran Academy", At most, 2);
	Switch("Switch136", Not Set);                         //DC1 아직 미인계 : 인계 완료 시 단락
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);   //미션2 진행 중 : 미션 단계 플래그 3
	Deaths("Player 8", "Terran Vulture", At least, 756); //미션2 인트로 시점 : 목표 표시 타이밍
Actions:
	Set Switch("Switch134", randomize); //DC1 인계 대상 무작위 재추첨
	Set Switch("Switch135", randomize);
	Preserve Trigger();
}

//----- (C) DefenseCenter1 give x4 (P1~P4) : 추첨 조합 + 생존 확인 시 1회 인계 -----//
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Academy", At least, 1);    //쉬움/보통
	Deaths("Player 8", "Terran Academy", At most, 2);
	Switch("Switch136", Not Set);                         //DC1 아직 미인계
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);   //미션2 진행 중
	Deaths("Player 8", "Terran Vulture", At least, 756); //미션2 인트로 시점
	Switch("Switch134", Not Set);                         //선택자 조합 : P1 (clear/clear)
	Switch("Switch135", Not Set);
	Command("Player 1", "Buildings", At least, 1);       //P1 생존 확인 : 죽었으면 발동X→재추첨
Actions:
	Set Invincibility("Player 9", "Buildings", "DefenseCenter1", disabled); //인계 시 무적 해제 : 방어선으로 활용
	Give Units to Player("Player 9", "Player 1", "Buildings", All, "DefenseCenter1"); //DC1 전체 건물 P1에게 통제권 이양
	Minimap Ping("DefenseCenter1");
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Display Text Message(Always Display, "<07>방어 거점 인계: <04>DefenseCenter1 구역의 방어 건물 통제권을 이양했습니다. 무적이 해제되었으니 방어선으로 활용하십시오.");
	Set Switch("Switch136", set); //DC1 인계 완료 : roll/give 중단
}
