//-----------------------------------------------------------------//
//  Mission 1 - outer colony attack penalty
//
//  If players bypass the first colony and attack P6/P7 during mission1,
//  destroyed outer-colony structures feed resources back into that colony.
//  If either outer colony's Hatchery/Lair/Hive collapses, both P6 and P7 wake up.
//  The first tech-core collapse also primes one large assault from that colony only.
//-----------------------------------------------------------------//

// Repeat the outer-colony intrusion warning at most once every 20 seconds.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Protoss Corsair", At least, 1);
Actions:
	Set Deaths("Player 8", "Protoss Corsair", Subtract, 1);
	Preserve Trigger();
}

Trigger("Player 1"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Protoss Corsair", Exactly, 0);
	Bring("Player 1", "Men", "P6 Alert", At least, 1);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Move Location("Player 1", "Men", "P6 Alert", "P6 Alert Ping");
	Minimap Ping("P6 Alert Ping");
