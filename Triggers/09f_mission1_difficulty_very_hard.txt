//-----------------------------------------------------------------//
//  Mission 1 difficulty adjustment - Very Hard
//
//  Player 8 / Terran Academy = 4 very hard.
//  These are extra additions on top of hard adjustment.
//-----------------------------------------------------------------//

// Wave 1 very hard reinforcement - added over hard.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
	Elapsed Time(At least, 210);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn4");
}

// Wave 2 very hard reinforcement - added over hard.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
	Elapsed Time(At least, 450);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn2");
}

// Wave 3 very hard reinforcement - reduced mixed-force tier over hard.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
	Elapsed Time(At least, 630);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn3");
	// 매우 어려움 누적 3차 병력의 10% 감축을 위해 중앙 히드라 증원 1기를 제외
	Create Unit("Player 5", "Zerg Lurker", 1, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Mutalisk", 1, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Scourge", 2, "P5 Spawn1");
}

// Colony destruction penalty: +5 seconds beyond hard (60 ticks after the -10% late-resistance reduction).
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
	Deaths("Player 5", "Zerg Hatchery", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 60);
	Preserve Trigger();
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
	Deaths("Player 5", "Zerg Lair", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 60);
	Preserve Trigger();
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
	Deaths("Player 5", "Zerg Hive", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 60);
	Preserve Trigger();
}

// Convert the easy 210s late-wave threshold into very hard 180s.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
	Deaths("Player 8", "Terran Wraith", At least, 2160);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 360);
	Preserve Trigger();
}
