//-----------------------------------------------------------------//
//  Mission 1 - dynamic building tracking and random all-out attack
//
//  Protoss Shuttle DC = Hunt location refresh timer (10s = 120)
//  Terran Barracks DC = P5 all-out attack timer (60s = 720)
//  Switch42/43 = P5 random Hunt target selector
//    clear/clear = P1, set/clear = P2, clear/set = P3, set/set = P4.
//  If a dead player is selected, the timer is not reset and rolls again.
//-----------------------------------------------------------------//

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Command("Player 1", "Buildings", At least, 1);
	Deaths("Player 8", "Protoss Shuttle", At least, 120);
Actions:
	Move Location("Player 1", "Buildings", "Anywhere", "P1 Hunt");
	Preserve Trigger();
}

Actions:

}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Barracks", At least, 720);
Actions:
	Set Switch("Switch42", randomize);
	Set Switch("Switch43", randomize);
	Preserve Trigger();
}
