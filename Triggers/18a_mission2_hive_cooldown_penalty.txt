//-----------------------------------------------------------------//
//  Mission 2 - hatchery tech loss accelerates main assaults
//
//  Hatchery/Lair/Hive kills reduce that colony's own main assault
//  cooldown by 100s and grant players +100/+200/+300 minerals.
//  12 DC tick = 1 second, so 100s = 1200.
//-----------------------------------------------------------------//

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 6", "Zerg Hatchery", At least, 1);
Actions:
	Display Text Message(Always Display, "<07>작전 보상: <15>갈색 저그<04> 해처리 파괴 보상. 미네랄 +100");
	Display Text Message(Always Display, "<08>경고: <04>군락이 격분했습니다. 해당 군락의 다음 주공세가 앞당겨집니다.");
	Preserve Trigger();
}

Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 6", "Zerg Hatchery", At least, 1);
Actions:
	Set Deaths("Player 8", "Zerg Zergling", Add, 1200);
	Set Resources("Player 1", Add, 100, ore);
	Set Resources("Player 2", Add, 100, ore);
	Set Resources("Player 3", Add, 100, ore);
	Set Resources("Player 4", Add, 100, ore);
	Set Deaths("Player 6", "Zerg Hatchery", Subtract, 1);
	Preserve Trigger();
}
