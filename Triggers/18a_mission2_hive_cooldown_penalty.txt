//-----------------------------------------------------------------//
//  Mission 2 - hatchery tech loss accelerates main assaults
//
//  Hatchery/Lair/Hive kills reduce that colony's own main assault
//  cooldown by difficulty and grant players minerals/gas.
//  12 DC tick = 1 second: easy 97.5s = 1170, normal 371.25s = 4455, hard 382.5s = 4590, very hard 427.5s = 5130.
//  Hard also accelerates the opposite colony by 191.25s = 2295; very hard accelerates it by the full 427.5s = 5130.
//-----------------------------------------------------------------//

// 실제 가속 조건과 같은 가드로 사건을 대기열에 적재한다.
// 해당 사망 카운터는 아래 가속/보상 트리거가 사이클당 1개씩 소비하므로 동시 파괴 수도 보존된다.
Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3); //미션2 진행 중 : 생산 구조 파괴 가속 활성 구간
	Deaths("Player 8", "Terran Academy", At least, 1); //난이도 선택 완료 : 쉬움~매우 어려움
	Deaths("Player 8", "Terran Academy", At most, 4); //유효 난이도 범위 : 실제 가속 트리거와 일치
	Switch("Switch32", Not Set); //P7 군락 생존 중 : P6 생산 구조 파괴가 실제 공세를 가속
	Switch("Switch158", Not Set); //P6 최종 절규 공세 종료 전 : P6 주공세 가속 잠금 전
	Deaths("Player 6", "Zerg Hatchery", At least, 1); //P6 해처리 파괴 사건 대기
Actions:
	Set Deaths("Player 8", "Terran Covert Ops", Add, 1); //P6 해처리 분노 후속 메시지 1건 적재
	Preserve Trigger();
}

//  Accelerated tech-loss follow-up queue
//
//  P6/P7 acceleration triggers enqueue the destroyed structure type.
//  Player 8 publishes one reward code after human-player triggers have

//  then clears it after the red reaction. Simultaneous kills therefore show
//  as reward -> rage pairs without same-frame message overwrites.

	Bring("Player 6", "Zerg Hive", "P6 MainBase", At most, 0);

	Set Deaths("Player 8", "Terran Covert Ops", Set To, 0); //P6 해처리 후속 메시지 대기열 폐기

	Set Deaths("Player 8", "Protoss Nexus", Set To, 0); //P7 레어 후속 메시지 대기열 폐기
	Set Deaths("Player 8", "Terran Command Center", Set To, 0); //P7 하이브 후속 메시지 대기열 폐기
