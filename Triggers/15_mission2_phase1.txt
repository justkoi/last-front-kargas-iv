// 페이즈1 시작 브리핑 (Valkyrie 2 진입 직후 1회)
Trigger("All Players"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
Actions:
	Set Mission Objectives("미션 2: 통신 두절의 진실");
	Display Text Message(Always Display, " ");
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "  [미션 2 개시] 통신 두절의 진실");
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "[노바] 사령관, 북쪽 중앙에서 UED 암호 신호 감지.");
	Display Text Message(Always Display, "[노바] 그런데 패턴이 이상해요. 정확히 7.3초 주기로 반복돼요.");
	Display Text Message(Always Display, "[노바] 살아있는 송신자가 아닐 수 있습니다. 자동 송출일 가능성이 높아요.");
	Display Text Message(Always Display, "[호크] 함정일 수도 있다는 거군. 그래도 출처는 봐야 한다.");
	Display Text Message(Always Display, "[호크] 정비 5분. 본진 보강하고 진군 준비해라.");
	Set Countdown Timer(Set To, 300);
}

// 페이즈1 150초 경과 - 중간 알림
Trigger("All Players"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
	Countdown Timer(At most, 120);
Actions:
	Display Text Message(Always Display, "[호크] 정비 3분 경과. 2분 남았다.");
}

// 페이즈1 종료 - 페이즈2 진입 (Valkyrie 2→3, Vulture 0 리셋)
Trigger("All Players"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
	Countdown Timer(At most, 1);
Actions:
	Set Deaths("Player 8", "Terran Valkyrie", Set To, 3);
	Set Deaths("Player 8", "Terran Vulture", Set To, 0);
}

// P5 Lab Vision 영역 - 미션2 시작 시 일시적 정찰
// Map Revealer 잠깐 띄워서 fog "탐색됨" 상태로 만든 뒤 제거.
// → 지형은 보이지만 정상 fog 메커니즘 유지(전투 시 시야 필요).
// 페이즈1 진입 시 스폰, 5초 후(Vulture ≥ 60틱) 제거.
Trigger("Player 1"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
Actions:
	Create Unit("Player 1", "Map Revealer", 1, "P5 Lab Vision");
}
Trigger("Player 2"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
Actions:
	Create Unit("Player 2", "Map Revealer", 1, "P5 Lab Vision");
}
Trigger("Player 3"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
Actions:
	Create Unit("Player 3", "Map Revealer", 1, "P5 Lab Vision");
}
Trigger("Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
Actions:
	Create Unit("Player 4", "Map Revealer", 1, "P5 Lab Vision");
}

// 5초 후 Map Revealer 제거 (Vulture DC ≥ 60틱)
Trigger("All Players"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
	Deaths("Player 8", "Terran Vulture", At least, 60);
	Deaths("Player 8", "Terran Vulture", At most, 84);
Actions:
	Remove Unit("Player 1", "Map Revealer");
	Remove Unit("Player 2", "Map Revealer");
	Remove Unit("Player 3", "Map Revealer");
	Remove Unit("Player 4", "Map Revealer");
}
