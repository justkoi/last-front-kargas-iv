// 페이즈1 시작 브리핑 (Valkyrie 2 진입 직후 1회) — 헤더 + 오브젝티브 + 타이머
// Armory DC를 대사 진행 카운터로 사용 (1=시작, 2~5=대사 단계)
Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 2);
	Deaths("Player 8", "Terran Vulture", At least, 120);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 2500);
	Set Mission Objectives("[미션 2 - 페이즈 1] 정비 단계\n\n▶ 목표: 2분간 부대 정비 및 재편성\n▶ 클리어 조건: 카운트다운 타이머 종료 (자동 진행)\n▶ 시간: 120초\n▶ 다음 단계: 페이즈 2 신호원 추적 시작\n▶ 팁: 자원 비축, 부대 구성, 본진 재정비 권장");
	Display Text Message(Always Display, " ");
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "  [미션 2 개시] 통신 두절의 진실");
	Display Text Message(Always Display, "==========================================");
	Set Countdown Timer(Set To, 120);
	Set Deaths("Player 8", "Terran Armory", Set To, 1);
}
