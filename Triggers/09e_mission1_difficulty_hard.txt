//-----------------------------------------------------------------//
//  Mission 1 difficulty adjustment - Hard
//
//  Player 8 / Terran Academy = 3 hard, 4 very hard.
//  These are extra additions on top of normal adjustment.
//-----------------------------------------------------------------//

// Wave 1 hard reinforcement - +2 Zerglings / +1 Hydralisk over normal.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 3);
	Elapsed Time(At least, 210);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn3");
}

// Wave 2 hard reinforcement - added over normal.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 3);
	Elapsed Time(At least, 450);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn2");
}

// Wave 3 hard reinforcement - reduced mixed-force tier over normal.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 3);
	Elapsed Time(At least, 630);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn1");
	// 어려움 누적 3차 병력의 10% 감축을 위해 측면 저글링 증원 1기를 제외
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Lurker", 1, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Mutalisk", 1, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Scourge", 2, "P5 Spawn4");
}

// Colony destruction penalty: +5.5 seconds beyond normal (66 = +10% over the old 60).
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 3);
	Deaths("Player 5", "Zerg Hatchery", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 66);
	Preserve Trigger();
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 3);
	Deaths("Player 5", "Zerg Lair", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 66);
	Preserve Trigger();
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 3);
	Deaths("Player 5", "Zerg Hive", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 66);
	Preserve Trigger();
}

// Convert the easy 210s late-wave threshold into hard 190s.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 3);
	Deaths("Player 8", "Terran Wraith", At least, 2280);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 240);
	Preserve Trigger();
}
