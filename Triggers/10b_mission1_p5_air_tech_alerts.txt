Trigger("Player 1"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Command("Player 5", "Zerg Mutalisk", At least, 1);
	Switch("Switch97", Not Set);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Minimap Ping("P5 Base");
	Display Text Message(Always Display, "<1A>노바: <04>뮤탈리스크 출현 확인. 기동성이 좋으니 기습에 대비하는 것이 좋겠습니다.");
	Set Switch("Switch97", set);
	Preserve Trigger();
}
