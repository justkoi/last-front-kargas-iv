//  [점령] 비콘 진입 시 Probe=1 + 타이머 정지. 점거 중일 때 Zealot 누적 증가.
//         총 누적 5분(3600틱) = 100% 점령 완료.
//         10%마다 진행도 알림, 20%마다 P6 반격조 파병.
//         60% 시점 = 거의 모든 조합 동원하는 대규모 총공세 (전복 가능).
//  [반복 침투] 타이머 0 + 점령 미시작 시 반복 침투 플래그 ON,
//              비콘 진입 시 반복 플래그 OFF.
//
//  마일스톤 1회성 침투는 17b_milestone_infiltration.txt에 분리됨.
//
//  Protoss Probe DC   : 비콘 점거 토글 (0=비어있음, 1=플레이어 점거 중)
//  Protoss Zealot DC  : 누적 점령도 (0~3600, 12 tick = 1초, 3600 = 5분 = 100%)
//  Terran Wraith DC   : 반복 침투 쿨다운 (720틱 = 60s)
//  Terran Dropship DC : 반복 침투 플래그 (0=OFF, 1=ON)
//-----------------------------------------------------------------//

// 페이즈2 진입 시 점령/침투 관련 DC 0 리셋
Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Vulture", At most, 12);
Actions:
	Set Deaths("Player 8", "Protoss Probe", Set To, 0);
	Set Deaths("Player 8", "Protoss Zealot", Set To, 0);
	Set Deaths("Player 8", "Terran Wraith", Set To, 0);
	Set Deaths("Player 8", "Terran Dropship", Set To, 0);
}

// 첫 비콘 진입 - 1회성 인트로 다이얼로그
Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Probe", Exactly, 0);
	Bring("Foes", "Men", "P5 Lab Beacon", At least, 1);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 2000);
	Display Text Message(Always Display, "[신호원 도달] 폐기지 단말기 확보.");
	Display Text Message(Always Display, "[노바] 단말기 데이터 추출 시작. 누적 5분간 비콘 사수 필요.");
	Display Text Message(Always Display, "[노바] 추출 진행에 따라 변종 반격이 점점 강해집니다.");
	Set Mission Objectives("미션 2 [추출 중]: 비콘 누적 점거 5분 = 100%");
	Set Deaths("Player 8", "Terran Dropship", Set To, 0);
}

// 점거 토글 ON - 비콘 점거 시 Probe=1 + 타이머 정지
Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Bring("Foes", "Men", "P5 Lab Beacon", At least, 1);
Actions:
	Pause Timer();
	Set Deaths("Player 8", "Protoss Probe", Set To, 1);
	Preserve Trigger();
}

// 점거 토글 OFF - 비콘 비면 Probe=0 + 타이머 재개
Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Bring("Foes", "Men", "P5 Lab Beacon", At most, 0);
Actions:
	Unpause Timer();
	Set Deaths("Player 8", "Protoss Probe", Set To, 0);
	Preserve Trigger();
}

// 누적 카운터 증가 - Probe=1일 때만, 매 트리거 사이클(12회/초)
Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Probe", Exactly, 1);
	Deaths("Player 8", "Protoss Zealot", At most, 3599);
Actions:
	Set Deaths("Player 8", "Protoss Zealot", Add, 1);
	Preserve Trigger();
}
