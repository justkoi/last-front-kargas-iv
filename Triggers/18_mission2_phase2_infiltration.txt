//-----------------------------------------------------------------//
//  미션 2 페이즈 2 - 비콘 점령 메커니즘 + 반복 침투 (P6)
//  [점령] 비콘 진입 → 타이머 영구정지, 60초 쿨다운마다 P6 반격조 스폰,
//         반격 1회당 점령도 +20%, 5회 = 100% → 미션2 클리어
//         (in-fiction: 단말기 데이터 추출, 5단계 진행률)
//  [반복 침투] 타이머 0 + 점령 미시작 시 반복 침투 플래그 ON,
//              비콘 진입 시 반복 플래그 OFF
//
//  마일스톤 1회성 침투는 17b_milestone_infiltration.txt에 분리됨.
//
//  Protoss Probe DC   : 점령 시작 플래그 (0=미시작, 1=시작됨)
//  Protoss Zealot DC  : 점령 단계 (0~5)
//  Terran Wraith DC   : 반격/반복침투 쿨다운 (360틱 = 30s)
//  Terran Dropship DC : 반복 침투 플래그 (0=OFF, 1=ON)
//-----------------------------------------------------------------//

// 페이즈2 진입 시 점령/침투 관련 DC 0 리셋
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Vulture", At most, 12);
Actions:
	Set Deaths("Player 8", "Protoss Probe", Set To, 0);
	Set Deaths("Player 8", "Protoss Zealot", Set To, 0);
	Set Deaths("Player 8", "Terran Wraith", Set To, 0);
	Set Deaths("Player 8", "Terran Dropship", Set To, 0);
}

// 첫 비콘 진입 - 1회성 인트로 다이얼로그 + 반복 침투 OFF
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Probe", Exactly, 0);
	Bring("Foes", "Men", "P5 Lab Beacon", At least, 1);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 2000);
	Display Text Message(Always Display, "[신호원 도달] 폐기지 단말기 확보.");
	Display Text Message(Always Display, "[노바] 단말기 데이터 추출 시작. 약 5분 걸려요.");
	Display Text Message(Always Display, "[노바] 그동안 남쪽 변종 무리가 추출을 막을 거예요. 5번 격퇴하면 추출 완료.");
	Set Mission Objectives("미션 2 [추출 중]: 변종 반격 5회 격퇴 (데이터 0/100%)");
	Set Deaths("Player 8", "Terran Dropship", Set To, 0);
}

// 비콘 점거 토글 ON - 비콘에 적(=플레이어) 있으면 Probe=1 + 타이머 정지
// (인트로와 동시 발동, Preserve로 재진입 대응)
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Probe", Exactly, 0);
	Bring("Foes", "Men", "P5 Lab Beacon", At least, 1);
Actions:
	Pause Timer();
	Set Deaths("Player 8", "Protoss Probe", Set To, 1);
	Preserve Trigger();
}

// 비콘 점거 토글 OFF - 비콘 비면 Probe=0 + 타이머 재개
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Probe", Exactly, 1);
	Bring("Foes", "Men", "P5 Lab Beacon", At most, 0);
Actions:
	Unpause Timer();
	Set Deaths("Player 8", "Protoss Probe", Set To, 0);
	Display Text Message(Always Display, "[경보] 비콘 비었습니다. 데이터 추출 일시중단, 타이머 재개.");
	Preserve Trigger();
}

// 반격 스폰 - 점령 시작됨 + 비콘 점거 중 + 쿨다운 완료 + 미완료
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Probe", Exactly, 1);
	Deaths("Player 8", "Protoss Zealot", At most, 4);
	Deaths("Player 8", "Terran Wraith", At least, 360);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Set Deaths("Player 8", "Terran Wraith", Set To, 0);
	Set Deaths("Player 8", "Protoss Zealot", Add, 1);
	Create Unit("Player 6", "Zerg Zergling", 30, "P6 South");
	Create Unit("Player 6", "Zerg Hydralisk", 18, "P6 South");
	Create Unit("Player 6", "Zerg Lurker", 4, "P6 South");
	Order("Player 6", "Zerg Zergling", "P6 South", "P5 Lab Beacon", Patrol);
	Order("Player 6", "Zerg Hydralisk", "P6 South", "P5 Lab Beacon", Patrol);
	Order("Player 6", "Zerg Lurker", "P6 South", "P5 Lab Beacon", Patrol);
	Display Text Message(Always Display, "[경보] 변종 반격조 진격! [데이터 +20%]");
	Preserve Trigger();
}
