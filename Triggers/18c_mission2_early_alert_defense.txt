//  Mission 2 - early P6/P7 alert defense

//  During the first 3 minutes of mission2, each colony gains at most

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Vulture", At most, 2159);
	Switch("Switch59", Not Set);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 1);
	Preserve Trigger();
}
