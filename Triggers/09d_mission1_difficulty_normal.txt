//-----------------------------------------------------------------//
//  Mission 1 difficulty adjustment - Normal
//
//  Player 8 / Terran Academy = 2 normal, 3 hard, 4 very hard.
//  Normal reinforcement is the first cumulative difficulty layer.
//-----------------------------------------------------------------//

// Wave 1 normal reinforcement - +1 Zergling / +1 Hydralisk over easy after the 10% wave reduction.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Elapsed Time(At least, 210);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2"); //보통 이상 1차 증원 : 중앙 경로 저글링 보강
	// 보통 이상 누적 병력의 약 10% 감축을 위해 1차 중앙 저글링 증원 1기를 제외
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn2"); //보통 이상 1차 대공 화력 보강
}

// Wave 2 normal reinforcement - +1 Zergling / +1 Hydralisk over easy after the 10% wave reduction.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Elapsed Time(At least, 450);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	// 보통 이상 누적 병력의 약 10% 감축을 위해 2차 중앙 저글링 증원 1기를 제외
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn4");
}

// Wave 3 normal reinforcement - reduced mixed-force tier over easy.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Elapsed Time(At least, 630);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	// 보통 이상 누적 병력의 약 10% 감축을 위해 3차 중앙 저글링 증원 1기를 제외
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Lurker", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Mutalisk", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Scourge", 2, "P5 Spawn3");
}

// Colony destruction penalty: +5 seconds beyond easy (60 ticks after the -10% late-resistance reduction).
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Deaths("Player 5", "Zerg Hatchery", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 60);
	Preserve Trigger();
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Deaths("Player 5", "Zerg Lair", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 60);
	Preserve Trigger();
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Deaths("Player 5", "Zerg Hive", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 60);
	Preserve Trigger();
}

// Convert the easy 210s late-wave threshold into normal 200s.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 2);
	Deaths("Player 8", "Terran Wraith", At least, 2400);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 120);
	Preserve Trigger();
}
