//-----------------------------------------------------------------//
//  Mission 1 difficulty adjustment - Normal
//
//  Player 8 / Terran Academy = 2 normal, 3 hard.
//  Normal adjustment also applies to hard unless noted otherwise.
//-----------------------------------------------------------------//

// Wave 2 normal reinforcement - 8:00 launch group.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Elapsed Time(At least, 450);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn4");
}

// Wave 3 normal reinforcement - 11:00 launch group.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Elapsed Time(At least, 630);
	Command("Player 5", "Buildings", At least, 1);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Lurker", 1, "P5 Spawn4");
}

// Convert the easy 210s late-wave threshold into normal 200s.
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", Exactly, 2);
	Deaths("Player 8", "Terran Wraith", At least, 2400);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 120);
	Preserve Trigger();
}
