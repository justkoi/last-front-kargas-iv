// 1회차 - 실제 약 0:05 (Elapsed 8)
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1); //미션 1 진행 중일 때만 극초반 견제를 보냄
	Elapsed Time(At least, 8); //실제 5초에 가장 가까운 게임 시간(8 / 1.5 = 약 5.3초)
	Command("Player 5", "Buildings", At least, 1); //P5 군락이 살아 있을 때만 병력을 생성
Actions:
	Create Unit("Player 5", "Zerg Zergling", 3, "P5 Spawn1"); //극초반 근접 견제 3기 생성
	Create Unit("Player 5", "Zerg Hydralisk", 1, "P5 Spawn1"); //영웅과 공중 유닛을 압박할 원거리 병력 1기 생성
	Order("Player 5", "Zerg Zergling", "P5 Spawn1", "Player Destination", Patrol); //저글링을 플레이어 진영으로 즉시 공격 진군
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn1", "Player Destination", Patrol); //히드라리스크를 플레이어 진영으로 즉시 공격 진군
}
