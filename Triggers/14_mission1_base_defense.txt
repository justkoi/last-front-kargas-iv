//-----------------------------------------------------------------//
//  Base defense response
//
//  P5: Terran Marine DC, P6: Terran Armory DC, P7: Protoss Zealot DC
//  P5 cooldown: 10s = 120 ticks. P6/P7 cooldown: 6s = 72 ticks.
//
//  P5 uses P5 Area for detection and P5 Intruder as the moving target.
//  P6/P7 Area Ground/Air locations are map-side filtered. Each response
//  moves the single Intruder location onto the detected unit, then orders
//  the matching defenders to patrol there. Overlords in East Area are included
//  only during that colony's home-defense response.
//-----------------------------------------------------------------//

Trigger("Player 5"){
Conditions:
	Switch("Switch133", Not Set);
	Deaths("Player 8", "Terran Marine", At least, 120);
	Bring("Player 1", "Men", "P5 Area", At least, 1);
Actions:
	Move Location("Player 1", "Men", "P5 Area", "P5 Intruder");
	Set Deaths("Player 8", "Terran Marine", Set To, 0);
	Set Deaths("Player 8", "Terran Barracks", Set To, 540);
	Order("Player 5", "Zerg Zergling", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Hydralisk", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Lurker", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Mutalisk", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Scourge", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Ultralisk", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Guardian", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Devourer", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Queen", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Zerg Defiler", "Anywhere", "P5 Intruder", Patrol);
	Order("Player 5", "Infested Terran", "Anywhere", "P5 Intruder", Patrol);
	Preserve Trigger();
}
