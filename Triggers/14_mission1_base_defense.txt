//-----------------------------------------------------------------//
//  Base defense response
//
//  P5: Terran Marine DC, P6: Terran Armory DC, P7: Protoss Zealot DC
//  Cooldown: 90s = 1080 ticks.
//
//  Move Location does not accept "Foes" reliably in TrigEdit, so each
//  player intrusion is handled explicitly.
//-----------------------------------------------------------------//

Trigger("Player 5"){
Conditions:
	Bring("Player 1", "Men", "P5 Area", At least, 1);
	Deaths("Player 8", "Terran Marine", At least, 1080);
Actions:
	Set Deaths("Player 8", "Terran Marine", Set To, 0);
	Order("Player 5", "Zerg Zergling", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Zerg Hydralisk", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Zerg Lurker", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Zerg Mutalisk", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Zerg Scourge", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Zerg Ultralisk", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Zerg Guardian", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Zerg Devourer", "Anywhere", "P5 Area", Patrol);
	Order("Player 5", "Infested Terran", "Anywhere", "P5 Area", Patrol);
	Preserve Trigger();
}
