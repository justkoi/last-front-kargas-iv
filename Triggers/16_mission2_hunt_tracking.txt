//-----------------------------------------------------------------//
//  Mission-independent - dynamic building tracking and random all-out attack
//
//  Protoss Shuttle DC = Hunt location refresh timer (10s = 120)
//  Terran Firebat DC = P6 all-out attack timer (50s = 600)
//  Protoss Arbiter DC = P7 all-out attack timer (50s = 600)
//  Switch44/45 = P6 random Hunt target selector
//  Switch46/47 = P7 random Hunt target selector
//    clear/clear = P1, set/clear = P2, clear/set = P3, set/set = P4.
//  If a dead player is selected, the timer is not reset and rolls again.
//-----------------------------------------------------------------//

Trigger("Player 6"){
Conditions:
	Command("Player 1", "Buildings", At least, 1);
	Deaths("Player 8", "Protoss Shuttle", At least, 120);
Actions:
	Move Location("Player 1", "Buildings", "Anywhere", "P1 Hunt");
	Preserve Trigger();
}
