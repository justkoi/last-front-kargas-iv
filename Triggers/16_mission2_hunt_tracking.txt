//-----------------------------------------------------------------//
//  Mission-independent - dynamic building tracking and random all-out attack
//
//  Protoss Shuttle DC = Hunt location refresh timer (10s = 120)
//  Terran Firebat DC = P6 all-out attack timer (60s = 720)
//  Protoss Arbiter DC = P7 all-out attack timer (60s = 720)
//  Switch44/45 = P6 random Hunt target selector
//  Switch46/47 = P7 random Hunt target selector
//  Switch134 = P6 Westonia Hunt target-group selector
//  Switch135 = P7 Westonia Hunt target-group selector
//    clear/clear = P1, set/clear = P2, clear/set = P3, set/set = P4.
//  If Westonia group is selected but Westonia has no player buildings, the group selector is cleared and existing Anywhere targeting is used.
//  If a dead player is selected in Anywhere mode, the timer is not reset and rolls again.
//-----------------------------------------------------------------//
