// Mission 1 late-wave cooldown starts after 15 minutes.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Elapsed Time(At least, 900);
	Command("Player 5", "Zerg Cerebrate", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Wraith", Add, 1);
	Preserve Trigger();
}

// Mission 2 main assault cooldowns advance during mission2.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Vulture", At least, 756);
Actions:
	Set Deaths("Player 8", "Zerg Zergling", Add, 1);
	Set Deaths("Player 8", "Zerg Hydralisk", Add, 1);
	Preserve Trigger();
}

// Mission 2 P6 MiniBase reveal timer starts after the first P6 main assault begins spawning.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Ghost", At least, 1);
	Switch("Switch48", Not Set);
Actions:
	Set Deaths("Player 8", "Terran Ghost", Add, 1);
	Preserve Trigger();
}
