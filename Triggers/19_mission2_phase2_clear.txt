//-----------------------------------------------------------------//
//  미션 2 페이즈 2 - 데이터 추출 완료 / 클리어
//  Zealot DC ≥ 3600 (누적 5분 점거 완료) → Valkyrie 3→4
//  카르츠 사전 녹화 메시지 → 미션 3 좌표 잠금 해제
//-----------------------------------------------------------------//

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Zealot", At least, 3600);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 2500);
	Pause Timer();
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "  [데이터 추출 완료] 신호원 단말기 분석 종료");
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "[노바] 추출 완료. 사령관... 이건 UED 잔당이 아니에요.");
	Display Text Message(Always Display, "[노바] 신호는 처음부터 미끼였어요. 누군가 우릴 끌어들였습니다.");
	Set Deaths("Player 8", "Terran Valkyrie", Set To, 4);
	Set Deaths("Player 8", "Terran Vulture", Set To, 0);
}

// 카르츠의 사전 녹화 메시지 (점령 5초 후)
Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 4);
	Deaths("Player 8", "Terran Vulture", At least, 60);
	Deaths("Player 8", "Terran Vulture", At most, 84);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 1500);
	Display Text Message(Always Display, "[?? - 사전 녹화] 축하한다, 선봉대.");
	Display Text Message(Always Display, "[??] 너희는 늦었다. 그리고 너희 존재가 내 다음 데이터다.");
	Display Text Message(Always Display, "[노바] ...박사 셀렌 카르츠. 도미니언 출신 변종 연구자.");
	Display Text Message(Always Display, "[노바] 데이터에 다음 좌표 - 북서쪽 프로토스 유적. 카운트다운이 걸려 있어요.");
	Display Text Message(Always Display, "[호크] 그놈은 거기서 뭔가를 끝내려 한다. 우린 그 전에 도달해야 해.");
	Set Mission Objectives("[미션 2 완료] 통신 두절의 진실\n\n▶ 신호원 데이터 추출 완료\n▶ 적 식별: 박사 셀렌 카르츠 (도미니언 출신 변종 연구자)\n▶ 다음 좌표: 북서쪽 프로토스 유적\n▶ 카운트다운 진행 중 — 다음 작전 준비");
}

// 미션 2 완료 - 임시 Victory (미션 3 작성 전까지)
Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 4);
	Deaths("Player 8", "Terran Vulture", At least, 180);

	Display Text Message(Always Display, "  [미션 2 완료] - 수고하셨습니다! 미션 3 작성 예정");

	Wait(1000);
	Victory();
}
