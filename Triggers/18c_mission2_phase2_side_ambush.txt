//-----------------------------------------------------------------//
//  Mission 2 Phase 2 - Side Event: Ambush
//  Protoss Dragoon DC   : ambush state (0=ready, 1=alerted, 2=ambush spawned)
//  Terran Goliath DC    : alert dialogue timer, reset when alert starts
//-----------------------------------------------------------------//

Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Vulture", At most, 12);
Actions:
	Set Deaths("Player 8", "Protoss Dragoon", Set To, 0);
}

// First entry warning - 10 or more player units enter the suspicious approach.
Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Dragoon", Exactly, 0);
	Bring("Foes", "Men", "Ambush Alert", At least, 10);
Actions:
	Set Deaths("Player 8", "Protoss Dragoon", Set To, 1);
	Set Deaths("Player 8", "Terran Goliath", Set To, 0);
}

// Test-friendly fallback: entering the midpoint directly also arms the ambush.

	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Protoss Dragoon", Exactly, 0);
	Bring("Foes", "Men", "Ambush Triggers", At least, 10);
Actions:
	Set Deaths("Player 8", "Protoss Dragoon", Set To, 1);
	Set Deaths("Player 8", "Terran Goliath", Set To, 0);
}
