//-----------------------------------------------------------------//
//  저그 시작 보너스 + AI
//  P6/P7은 미션1 동안 자원을 0으로 묶고, 미션2부터 활성화한다.
//-----------------------------------------------------------------//

Trigger("Player 5"){
Conditions:
	Elapsed Time(At most, 1);
Actions:
	Set Resources("Player 5", Set To, 999999, ore);
	Set Resources("Player 5", Set To, 999999, gas);
}

// 미션1 동안 P6/P7은 잠복 상태로 둔다.
Trigger("Player 8"){
Conditions:
	Elapsed Time(At most, 2);
Actions:
	Set Switch("Switch24", clear);
	Set Switch("Switch25", clear);
	Set Switch("Switch55", clear);
	Set Switch("Switch56", clear);
	Set Switch("Switch58", clear);
	Set Switch("Switch59", clear);
	Set Switch("Switch60", clear);
	Set Switch("Switch13", clear);
	Set Switch("Switch35", clear);
	Set Switch("Switch36", clear);
	Set Deaths("Player 8", "Zerg Creep Colony", Set To, 0);
	Set Deaths("Player 8", "Terran Wraith", Set To, 0);
	Set Deaths("Player 8", "Terran Dropship", Set To, 0);
	Set Resources("Player 6", Set To, 0, ore);
	Set Resources("Player 6", Set To, 0, gas);
	Set Resources("Player 7", Set To, 0, ore);
	Set Resources("Player 7", Set To, 0, gas);
}

// 각 저그 진영의 기본 AI 시작
// P5 AI는 난이도 확정(Terran Academy >= 1) 이후에만 가동한다.
Trigger("Player 5"){
Conditions:
	Elapsed Time(At least, 330);
	Deaths("Player 8", "Terran Academy", At least, 1);
Actions:
	Run AI Script At Location("ZSUx", "P5 Base");
}

// P6/P7은 Switch55/56이 켜질 때 자원을 먼저 받고 AI를 시작한다.

// Keep outer colonies resource-starved during mission1 until that colony is attacked.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Switch("Switch24", Not Set);
Actions:
	Set Resources("Player 6", Set To, 0, ore);
	Set Resources("Player 6", Set To, 0, gas);
	Preserve Trigger();
}

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Switch("Switch25", Not Set);
Actions:
	Set Resources("Player 7", Set To, 0, ore);
	Set Resources("Player 7", Set To, 0, gas);
	Preserve Trigger();
}

// P6/P7 AI activation is requested at mission2 entry in 12_mission1_to_mission2_transition.txt.

Trigger("Player 6"){
Conditions:
	Switch("Switch55", Set);
Actions:
	Set Resources("Player 6", Set To, 999999, ore);
	Set Resources("Player 6", Set To, 999999, gas);
	Create Unit("Player 6", "Zerg Overlord", 10, "P6 Base");
	Run AI Script At Location("ZSUx", "P6 Base");
}

Trigger("Player 7"){
Conditions:
	Switch("Switch56", Set);
Actions:
	Set Resources("Player 7", Set To, 999999, ore);
	Set Resources("Player 7", Set To, 999999, gas);
	Create Unit("Player 7", "Zerg Overlord", 10, "P7 Base");
	Run AI Script At Location("ZSUx", "P7 Base");
}
