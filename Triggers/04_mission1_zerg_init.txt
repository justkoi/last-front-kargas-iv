// 각 저그 진영의 기본 AI 시작
// P5 AI는 난이도 확정(Terran Academy >= 1) 이후에만 가동한다.
// 난이도별 Elapsed Time 활성화 시점: 쉬움(1)=240, 보통(2)=190, 어려움(3)=175, 매우 어려움(4)=160.
// Fastest 실제 플레이 시간 환산: 약 2:40 / 2:07 / 1:57 / 1:47.
Trigger("Player 5"){
Conditions:
	Elapsed Time(At least, 240);
	Deaths("Player 8", "Terran Academy", Exactly, 1);
Actions:
	Run AI Script At Location("ZSUx", "P5 Base");
}

Trigger("Player 5"){
Conditions:
	Elapsed Time(At least, 190);
	Deaths("Player 8", "Terran Academy", Exactly, 2);
Actions:
	Run AI Script At Location("ZSUx", "P5 Base");
}

Trigger("Player 5"){
Conditions:
	Elapsed Time(At least, 175);
	Deaths("Player 8", "Terran Academy", Exactly, 3);
Actions:
	Run AI Script At Location("ZSUx", "P5 Base");
}

Trigger("Player 5"){
Conditions:
	Elapsed Time(At least, 160);
	Deaths("Player 8", "Terran Academy", Exactly, 4);
Actions:
	Run AI Script At Location("ZSUx", "P5 Base");
}

// P6/P7은 Switch55/56이 켜질 때 자원을 먼저 받고 AI를 시작한다.

// Keep outer colonies resource-starved during mission1 until that colony is attacked.
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Switch("Switch24", Not Set);
Actions:
	Set Resources("Player 6", Set To, 0, ore);
	Set Resources("Player 6", Set To, 0, gas);
	Preserve Trigger();
}

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Switch("Switch25", Not Set);
Actions:
	Set Resources("Player 7", Set To, 0, ore);
	Set Resources("Player 7", Set To, 0, gas);
	Preserve Trigger();
}

// P6/P7 AI activation is normally requested at the mission2 title timing (Vulture 756)
// in 12_mission1_to_mission2_transition.txt. Attacking an outer colony in mission1 can request it early.

Trigger("Player 6"){
Conditions:
	Switch("Switch55", Set);
Actions:
	Set Resources("Player 6", Set To, 999999, ore);
	Set Resources("Player 6", Set To, 999999, gas);
	Create Unit("Player 6", "Zerg Overlord", 10, "P6 Base");
	Run AI Script At Location("ZSUx", "P6 Base");
}
