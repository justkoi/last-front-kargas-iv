//-----------------------------------------------------------------//
//  Overlord supply - independent colony timers
//  P5: 20s = 240 ticks. P6/P7: 10s = 120 ticks.
//  Each colony creates 1 Overlord only while it has at most 32 Overlords
//  and at least 1 Drone.
//-----------------------------------------------------------------//

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Battlecruiser", At least, 240);
	Command("Player 5", "Zerg Overlord", At most, 32);
	Command("Player 5", "Zerg Drone", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Battlecruiser", Set To, 0);
	Create Unit("Player 5", "Zerg Overlord", 1, "P5 Scout");
	Preserve Trigger();
}

Trigger("Player 6"){
Conditions:
	Deaths("Player 8", "Protoss Dragoon", At least, 120);
	Command("Player 6", "Zerg Overlord", At most, 32);
	Command("Player 6", "Zerg Drone", At least, 1);
Actions:
	Set Deaths("Player 8", "Protoss Dragoon", Set To, 0);
	Create Unit("Player 6", "Zerg Overlord", 1, "P6 Base");
	Preserve Trigger();
}

Trigger("Player 7"){
Conditions:
	Deaths("Player 8", "Protoss High Templar", At least, 120);
	Command("Player 7", "Zerg Overlord", At most, 32);
	Command("Player 7", "Zerg Drone", At least, 1);
Actions:
