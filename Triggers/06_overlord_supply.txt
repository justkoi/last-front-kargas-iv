Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Battlecruiser", At least, 360);
	Command("Player 5", "Zerg Overlord", At most, 19);
	Command("Player 5", "Zerg Drone", At least, 1);
Actions:
	Set Deaths("Player 8", "Terran Battlecruiser", Set To, 0);
	Create Unit("Player 5", "Zerg Overlord", 1, "P5 Base");
	Preserve Trigger();
}
