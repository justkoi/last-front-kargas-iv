//-----------------------------------------------------------------//
//  Mission 1 Colony Rage difficulty scaling
//
//  Additive Create + Hunt Order layers on top of 09c base batches (easy = 260 supply).
//  Fires in the same spawn tick window after each base batch switch latches.
//  Difficulty gate = Terran Academy cumulative deaths (1=easy, 2=normal, 3=hard, 4=very hard).
//  Layers stack: Normal fires L2; Hard fires L2+L3; Very Hard fires L2+L3+L4.
//    Normal    (Academy >= 2): +26 supply -> 286 total (+10% vs easy 260)
//    Hard      (Academy >= 3): +40 supply -> 300 total (+15% vs easy 260; exact +15%=299 is
//                              not reachable with 4 symmetric fronts, so 300=+15.4% is the nearest step)
//    Very Hard (Academy == 4): +52 supply -> 312 total (+20% vs easy 260)
//  12 DC ticks = 1 second.
//-----------------------------------------------------------------//

// L2 batch 1 addon (Normal): +2 Hydra, +1 Zergling per front (+10 supply total).
Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 8", "Terran Academy", At least, 2);
	Switch("Switch58", Set);
	Switch("Switch79", Set);
	Deaths("Player 8", "Protoss Dark Archon", At least, 360);
	Deaths("Player 8", "Protoss Dark Archon", At most, 372);
Actions:
	Create Unit("Player 5", "Zerg Hydralisk", 2, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Hydralisk", 2, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Hydralisk", 2, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Hydralisk", 2, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Zergling", 1, "P5 Spawn4");
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn1", "P1 Hunt", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn1", "P1 Hunt", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn2", "P2 Hunt", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn2", "P2 Hunt", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn3", "P3 Hunt", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn3", "P3 Hunt", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn4", "P4 Hunt", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn4", "P4 Hunt", Patrol);
}
