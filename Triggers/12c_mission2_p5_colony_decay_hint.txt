//-----------------------------------------------------------------//
//  Mission 2 - early P5 colony decay hint
//
//  Warns players only when they destroy orange Sunken/Spore Colonies
//  before the scripted 3-minute 20-second creep-root decay.
//-----------------------------------------------------------------//

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3);
	Deaths("Player 8", "Terran Vulture", At most, 12);
Actions:
	Set Deaths("Player 5", "Zerg Sunken Colony", Set To, 0);
	Set Deaths("Player 5", "Zerg Spore Colony", Set To, 0);
	Set Deaths("Player 5", "Terran Marine", Set To, 0);
	Set Deaths("Player 5", "Terran Firebat", Set To, 0);
	Set Switch("Switch109", clear);
	Set Switch("Switch110", clear);
	Set Switch("Switch111", clear);
	Set Switch("Switch112", clear);
	Set Switch("Switch113", clear);
	Set Switch("Switch114", clear);
	Set Switch("Switch115", clear);
	Set Switch("Switch116", clear);
	Preserve Trigger();
}

Conditions:

	Deaths("Player 8", "Terran Vulture", At most, 2399);

Actions:
	Set Deaths("Player 5", "Terran Marine", Add, 1);
	Set Deaths("Player 5", "Zerg Sunken Colony", Subtract, 1);
	Preserve Trigger();
}
