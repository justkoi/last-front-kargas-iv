//-----------------------------------------------------------------//
//  Mission 1 - P5 Hive death ground counterattack
//
//  Fires once when a P5 Hive death is detected, before the late-wave
//  reward trigger consumes the P5 Zerg Hive death counter.
//  Composition: 96 Zerglings + 32 Hydralisks
//  = 80 Zerg supply.
//-----------------------------------------------------------------//

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 5", "Zerg Hive", At least, 1);
	Switch("Switch58", Not Set);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 2500);
	Minimap Ping("P5 Home");
	Display Text Message(Always Display, "<1A>노바: <04>하이브 붕괴 확인. 잠깐... 군락 내부 신호가 역류합니다.");
	Display Text Message(Always Display, "<03>호크: <04>잔존 지상군이 한꺼번에 깨어난다. 전선을 유지해라.");
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 5", "Zerg Hive", At least, 1);
	Switch("Switch58", Not Set);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 24, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Hydralisk", 8, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Zergling", 24, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Hydralisk", 8, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 24, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Hydralisk", 8, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Zergling", 24, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Hydralisk", 8, "P5 Spawn4");
	Order("Player 5", "Zerg Zergling", "P5 Spawn1", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn1", "Player Destination", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn2", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn2", "Player Destination", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn3", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn3", "Player Destination", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn4", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn4", "Player Destination", Patrol);
	Set Deaths("Player 8", "Terran Barracks", Set To, 720);
	Set Switch("Switch58", set);
}
