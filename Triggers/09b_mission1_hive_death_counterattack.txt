//-----------------------------------------------------------------//
//  Mission 1 - P5 Hive death ground counterattack
//
//  Fires once when a P5 Hive death is detected, before the late-wave
//  reward trigger consumes the P5 Zerg Hive death counter.
//  Composition: 64 Zerglings + 16 Hydralisks + 2 Ultralisks + 2 Defilers
//  = 60 Zerg supply.
//-----------------------------------------------------------------//

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 5", "Zerg Hive", At least, 1);
	Switch("Switch58", Not Set);
Actions:
	Play WAV("sound\\Misc\\TRescue.wav", 2500);
	Minimap Ping("P5 Home");
	Display Text Message(Always Display, "<06>[비상 경보] <04>하이브의 최후 신호가 군락 전역에 퍼졌습니다. 잔존 지상군이 아군 본진으로 쇄도합니다.");
}

Trigger("Player 5"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 1);
	Deaths("Player 5", "Zerg Hive", At least, 1);
	Switch("Switch58", Not Set);
Actions:
	Create Unit("Player 5", "Zerg Zergling", 16, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Hydralisk", 4, "P5 Spawn1");
	Create Unit("Player 5", "Zerg Ultralisk", 1, "P5 Spawn1");
	Create Unit with Properties("Player 5", "Zerg Defiler", 1, "P5 Spawn1", 1);
	Create Unit("Player 5", "Zerg Zergling", 16, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Hydralisk", 4, "P5 Spawn2");
	Create Unit("Player 5", "Zerg Zergling", 16, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Hydralisk", 4, "P5 Spawn3");
	Create Unit("Player 5", "Zerg Ultralisk", 1, "P5 Spawn3");
	Create Unit with Properties("Player 5", "Zerg Defiler", 1, "P5 Spawn3", 1);
	Create Unit("Player 5", "Zerg Zergling", 16, "P5 Spawn4");
	Create Unit("Player 5", "Zerg Hydralisk", 4, "P5 Spawn4");
	Order("Player 5", "Zerg Zergling", "P5 Spawn1", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn1", "Player Destination", Patrol);
	Order("Player 5", "Zerg Ultralisk", "P5 Spawn1", "Player Destination", Patrol);
	Order("Player 5", "Zerg Defiler", "P5 Spawn1", "Player Destination", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn2", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn2", "Player Destination", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn3", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn3", "Player Destination", Patrol);
	Order("Player 5", "Zerg Ultralisk", "P5 Spawn3", "Player Destination", Patrol);
	Order("Player 5", "Zerg Defiler", "P5 Spawn3", "Player Destination", Patrol);
	Order("Player 5", "Zerg Zergling", "P5 Spawn4", "Player Destination", Patrol);
	Order("Player 5", "Zerg Hydralisk", "P5 Spawn4", "Player Destination", Patrol);
	Set Deaths("Player 8", "Terran Barracks", Set To, 720);
	Set Switch("Switch58", set);
}
