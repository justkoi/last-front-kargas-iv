//-----------------------------------------------------------------//
//  Mission 2 - offense window announcements (공격 창 안내)
//
//  주공세 스폰 시작 3분(2160틱) 후 = 웨이브가 방어선에서 정리될 즈음,
//  "군락이 비어 있는 지금이 진격 기회"를 알린다. 매 사이클 반복하되
//  4종 무작위 대사 바리에이션(호크/노바)으로 피로를 줄인다.
//  구체적 시간 수치는 난이도/지터/가속에 따라 달라지므로 언급하지 않는다.
//
//  타이머 DC: P6 = P8 Terran SCV, P7 = P8 Terran Missile Turret.
//  바리에이션 스위치: P6 = Switch70/155, P7 = Switch156/157 (2비트 = 4종).
//  발화 구조: P1~P4 표시 트리거가 DC Exactly 2161을 읽어 같은 사이클에
//  4명 모두 표시하고, 그 사이클 마지막에 P8 리셋 트리거가 DC를 0으로
//  되돌린다 (플레이어 트리거가 P8보다 먼저 평가되는 순서 이용).
//  12 DC tick = 1초.
//-----------------------------------------------------------------//

//==================== P6 (갈색 저그) ====================//

// P6 arm - 일반 스폰 시작 시 타이머 가동 + 대사 무작위 선택
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Zerg Lurker", Exactly, 1); //P6 주공세 상태 : 일반 웨이브 스폰 진행 중
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3); //미션 단계 : 미션2 진행 중
	Deaths("Player 8", "Terran SCV", Exactly, 0); //공격 창 타이머 : 미가동 상태에서만 1회 가동
Actions:
	Set Deaths("Player 8", "Terran SCV", Set To, 1); //타이머 가동 : 스폰 시작 시점부터 카운트
	Set Switch("Switch70", randomize); //대사 바리에이션 bit0 : 호크/노바 선택
	Set Switch("Switch155", randomize); //대사 바리에이션 bit1 : 문장 변형 선택
	Preserve Trigger();
}

// P6 tick - 매 사이클 +1, 2161에서 정지 (2160틱 = 180초)
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran SCV", At least, 1); //타이머 가동 중일 때만

// P6 reset - 발화 틱이 지나간 사이클 끝에 타이머 정리 (P1~P4 표시 후 실행됨)

	Preserve Trigger();
}
