//-----------------------------------------------------------------//
//  [TEST MODE] Trigger the normal mission 1 clear path
//
//  This does not set mission 2 directly. It prepares the same condition
//  used by Triggers/12_endgame.txt:
//    - Valkyrie DC = 1
//    - No P5 Hatchery/Lair/Hive remains at P5 Base
//
//  Then 12_endgame advances Valkyrie 1 -> 2, grants rewards, and starts
//  the normal mission 2 phase 1 flow.
//-----------------------------------------------------------------//

Trigger("Player 8"){
Conditions:
	Elapsed Time(At most, 1);
Actions:
	Set Deaths("Player 8", "Terran Valkyrie", Set To, 1);
	Set Deaths("Player 8", "Terran Vulture", Set To, 0);
	Remove Unit At Location("Player 5", "Zerg Hatchery", 64, "P5 Base");
	Remove Unit At Location("Player 5", "Zerg Lair", 64, "P5 Base");
	Remove Unit At Location("Player 5", "Zerg Hive", 64, "P5 Base");
}
