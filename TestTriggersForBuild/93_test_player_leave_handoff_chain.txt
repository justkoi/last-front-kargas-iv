//-----------------------------------------------------------------//
//  [TEST MODE] Solo chained player-leave handoff test
//
//  P8 emulates P2 and P3 as connected. P2 stops first, so production
//  logic transfers P2 to different-race P3. P3 then stops, transferring
//  both its original assets and the inherited P2 assets to real P1.
//
//  Timeline
//    3s~15s : P2 heartbeat
//    3s~25s : P3 heartbeat
//    16s+   : P2 -> P3
//    26s+   : P3 -> P1
//    Expected: P2 10000/10000 + P3 2000/2000 = P1 receives 12000/12000
//              before unrelated mission rewards or income are counted.
//
//  Test state: Player 8 / Gui Montag (Firebat)
//    0 = setup pending, 1 = P2/P3 assets ready
//-----------------------------------------------------------------//

//가상 P2 접속 유지 : 먼저 이탈하여 P3에게 자산을 넘김
Trigger("Player 8"){
Conditions:
	Elapsed Time(At least, 3);
	Elapsed Time(At most, 15);
Actions:
	Set Deaths("Player 2", "Zerg Beacon", Set To, 1);
	Preserve Trigger();
}

//가상 P3 접속 유지 : P2 자산을 받은 뒤 두 번째로 이탈
Trigger("Player 8"){
Conditions:
	Elapsed Time(At least, 3);
	Elapsed Time(At most, 25);
Actions:
	Set Deaths("Player 3", "Zerg Beacon", Set To, 1);
	Preserve Trigger();
}

//P1이 테란이면 P2도 테란, P3는 저그로 구성
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Gui Montag (Firebat)", Exactly, 0);
	Elapsed Time(At least, 5);
	Deaths("Player 1", "Terran Beacon", Exactly, 1);
Actions:
	Set Deaths("Player 2", "Terran Beacon", Set To, 1);
	Set Deaths("Player 3", "Terran Beacon", Set To, 2);
	Create Unit("Player 2", "Terran Command Center", 1, "P1 Start");
	Create Unit("Player 2", "Terran Supply Depot", 2, "P1 Start");
	Create Unit("Player 2", "Terran Marine", 12, "P1 Start");
	Create Unit("Player 3", "Zerg Hatchery", 1, "P1 Start");
	Create Unit("Player 3", "Zerg Overlord", 2, "P1 Start");
	Create Unit("Player 3", "Zerg Hydralisk", 8, "P1 Start");
	Set Resources("Player 2", Set To, 10000, ore);
	Set Resources("Player 2", Set To, 10000, gas);
	Set Resources("Player 3", Set To, 2000, ore);
	Set Resources("Player 3", Set To, 2000, gas);
	Set Deaths("Player 8", "Gui Montag (Firebat)", Set To, 1);
}

//P1이 저그이면 P2도 저그, P3는 테란으로 구성
Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Gui Montag (Firebat)", Exactly, 0);
	Elapsed Time(At least, 5);
	Deaths("Player 1", "Terran Beacon", Exactly, 2);
Actions:
	Set Deaths("Player 2", "Terran Beacon", Set To, 2);
	Set Deaths("Player 3", "Terran Beacon", Set To, 1);
	Create Unit("Player 2", "Zerg Hatchery", 1, "P1 Start");
	Create Unit("Player 2", "Zerg Overlord", 2, "P1 Start");
	Create Unit("Player 2", "Zerg Hydralisk", 12, "P1 Start");
	Create Unit("Player 3", "Terran Command Center", 1, "P1 Start");
	Create Unit("Player 3", "Terran Supply Depot", 2, "P1 Start");
	Create Unit("Player 3", "Terran Marine", 8, "P1 Start");
	Set Resources("Player 2", Set To, 10000, ore);
	Set Resources("Player 2", Set To, 10000, gas);
	Set Resources("Player 3", Set To, 2000, ore);
	Set Resources("Player 3", Set To, 2000, gas);
	Set Deaths("Player 8", "Gui Montag (Firebat)", Set To, 1);
}
