//-----------------------------------------------------------------//
//  [TEST MODE] Mission 2 main assault DC dashboard
//
//  Standard StarCraft leaderboards cannot read death counters directly.
//  This test dashboard mirrors the two mission 2 main-assault cooldown
//  DCs into custom score, then restores the source DCs in the same pass.
//
//  P6 custom score = Player 8 / Zerg Zergling DC.
//  P7 custom score = Player 8 / Zerg Hydralisk DC.
//  Temp DCs: Player 8 / Zerg Spawning Pool, Zerg Evolution Chamber.
//-----------------------------------------------------------------//

Trigger("Player 8"){
Conditions:
	Always();
Actions:
	Set Score("Player 6", Set To, 0, Custom);
	Set Score("Player 7", Set To, 0, Custom);
	Set Deaths("Player 8", "Zerg Spawning Pool", Set To, 0);
	Set Deaths("Player 8", "Zerg Evolution Chamber", Set To, 0);
	Preserve Trigger();
}

// P6 main assault timer binary copy: source DC -> Player 6 custom score.

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Zerg Zergling", At least, 32768);
Actions:
	Set Deaths("Player 8", "Zerg Zergling", Subtract, 32768);
	Set Deaths("Player 8", "Zerg Spawning Pool", Add, 32768);
	Set Score("Player 6", Add, 32768, Custom);
	Preserve Trigger();
