//스테이지1 셋업: 양방향 배급 (P1<->P2), 테스트 초기화 겸용
Trigger("Player 8"){
Conditions:
	Switch("Switch246", Not Set);
	Deaths("Player 8", "Zerg Ultralisk Cavern", Exactly, 0);
	Deaths("Player 8", "Zerg Queen", Exactly, 0); //배급 사이클 대기 중
Actions:
	Set Switch("Switch246", set); //테스트 1회 초기화 가드
	Set Switch("Switch177", set); //배급제 강제 활성화 (미션 게이트 우회)
	Set Switch("Switch218", set); //P1 등록 ON
	Set Switch("Switch219", set); //P2 등록 ON
	Set Switch("Switch220", clear); //P3 등록 OFF
	Set Switch("Switch221", clear); //P4 등록 OFF
	Set Resources("Player 1", Set To, 10000, ore);
	Set Resources("Player 1", Set To, 300, gas);
	Set Resources("Player 2", Set To, 500, ore);
	Set Resources("Player 2", Set To, 10000, gas);
	Set Resources("Player 3", Set To, 0, ore);
	Set Resources("Player 3", Set To, 0, gas);
	Set Resources("Player 4", Set To, 0, ore);
	Set Resources("Player 4", Set To, 0, gas);
	Set Deaths("Player 8", "Terran Supply Depot", Set To, 1439); //다음 사이클에 1440 도달, 즉시 배급 개시
	Set Deaths("Player 8", "Zerg Ultralisk Cavern", Set To, 1);
	Preserve Trigger();
}
