//-----------------------------------------------------------------//
//  [TEST MODE] Mission 2 P6/P7 waypoint Hunt relay debugger
//
//  Prints the relay gate that is currently blocking waypoint -> Hunt orders.

//-----------------------------------------------------------------//

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Zerg Queen", At least, 12); //디버그 출력 완료 : P1~P4 표시 단계가 본 뒤 다음 주기를 위해 초기화
Actions:
	Set Deaths("Player 8", "Zerg Queen", Set To, 0); //디버그 출력 타이머 초기화 : 같은 P8 단계에서 즉시 지워 표시가 막히지 않게 증가보다 먼저 실행
	Preserve Trigger();
}

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", At least, 1); //미션 진행 중 : 패널티 주공세 릴레이가 의미 있는 구간
	Deaths("Player 8", "Terran Valkyrie", At most, 3); //미션 클리어 전 : 미션1에서 시작된 릴레이도 추적
Actions:
	Set Deaths("Player 8", "Zerg Queen", Add, 1); //디버그 출력 타이머 : 12틱마다 상태 메시지 1회 출력
	Preserve Trigger();
}
