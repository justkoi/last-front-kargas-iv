//-----------------------------------------------------------------//
//  [TEST MODE] Disable hunt all-out attack orders
//
//  Purpose:
//    Test whether the Zerg AI behaves better when periodic hunt
//    all-out Order actions never fire.
//
//  Method:
//    Keep only the all-out attack timer DCs at 0. Dynamic Hunt
//    location tracking still runs, and other wave / defense systems
//    are not touched.
//
//  Disabled timer groups:
//    P5 Hunt     : Barracks
//    P6 Hunt     : Firebat
//    P7 Hunt     : Arbiter
//-----------------------------------------------------------------//

Trigger("Player 8"){
Conditions:
	Always();
Actions:
	Set Deaths("Player 8", "Terran Barracks", Set To, 0);
	Set Deaths("Player 8", "Terran Firebat", Set To, 0);
	Set Deaths("Player 8", "Protoss Arbiter", Set To, 0);
	Preserve Trigger();
}
