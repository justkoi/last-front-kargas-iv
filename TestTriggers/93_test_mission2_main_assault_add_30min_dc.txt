//-----------------------------------------------------------------//
//  [TEST MODE] Mission 2 main assault DC +30 minutes
//
//  30 minutes = 1800 seconds = 21600 DC ticks (12 tick = 1 second).
//  Adds that amount to the normal P6/P7 mission 2 main assault timers.
//  Does not spawn waves directly; production roll triggers consume it.
//
//  P6 timer: Player 8 / Zerg Zergling
//  P7 timer: Player 8 / Zerg Hydralisk
//
//  Fires once when mission 2 setup is initialized
//  (Valkyrie DC = 3 and Switch50 is set), so it stacks on top of the normal M2 init stagger.
//  Pair with 99_test_mission2_skip.txt for a quick mission 2 test build.
//
//  Move this file to TestTriggersForBuild/ when you want it included.
//-----------------------------------------------------------------//

Trigger("Player 1", "Player 2", "Player 3", "Player 4"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3); //미션2 진행 중 : 주공세 DC가 의미 있는 시점
	Switch("Switch50", Set); //미션2 주공세 초기화 완료 : 초기 스태거 값이 설정된 뒤
Actions:
	Display Text Message(Always Display, "<06>TEST: <04>미션2 주공세 DC +30분 누적 (Zergling/Hydralisk +21600 tick).");
}

Trigger("Player 8"){
Conditions:
	Deaths("Player 8", "Terran Valkyrie", Exactly, 3); //미션2 진행 중 : init 스태거 이후 누적
	Switch("Switch50", Set); //미션2 주공세 초기화 완료 : 초기 스태거 값이 설정된 뒤
Actions:
	Set Deaths("Player 8", "Zerg Zergling", Add, 21600);
	Set Deaths("Player 8", "Zerg Hydralisk", Add, 21600);
}
