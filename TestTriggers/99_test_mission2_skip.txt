//-----------------------------------------------------------------//
//  [TEST MODE] 미션 2부터 시작 - 미션 1 스킵
//
//  이 파일이 Triggers/에 있으면 테스트 모드. 정상 게임에선 빼야 함.
//  비활성화 방법:
//    A. 파일 이름을 .txt가 아닌 다른 확장자로 변경 (예: .txt.off)
//    B. 또는 이 파일을 Triggers/ 밖으로 이동
//
//  하는 일:
//   1) Valkyrie DC = 2 (미션2 페이즈1 강제 진입)
//   2) Vulture DC = 0 (페이즈 타이머 리셋)
//   3) 미션1 클리어 보상 자원 지급
//   4) Player 5 (미션1 저그)의 모든 유닛/건물 제거 (간섭 방지)
//-----------------------------------------------------------------//

Trigger("All Players"){
Conditions:
	Elapsed Time(At most, 1);
Actions:
	Set Deaths("Player 8", "Terran Valkyrie", Set To, 2);
	Set Deaths("Player 8", "Terran Vulture", Set To, 0);
	Set Resources("All Players", Add, 500, ore);
	Set Resources("All Players", Add, 200, gas);
	Display Text Message(Always Display, "==========================================");
	Display Text Message(Always Display, "  [TEST MODE] 미션 2부터 시작");
	Display Text Message(Always Display, "==========================================");
}

// 미션1 저그(P5) 잔존 유닛/건물 제거 - 테스트 간섭 방지
Trigger("Player 5"){
Conditions:
	Elapsed Time(At most, 2);
Actions:
	Kill Unit("Player 5", "Buildings");
	Kill Unit("Player 5", "Men");
}
